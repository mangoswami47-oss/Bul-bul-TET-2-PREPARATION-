function clean(value) {
 return String(value ?? "").trim();
}

function normalizeHeader(value) {
 return clean(value)
   .toLowerCase()
   .replace(/[\s_\-]+/g, "")
   .replace(/[.():]/g, "");
}

function getField(row, names) {
 const keys = Object.keys(row);

 for (const name of names) {
   const wanted = normalizeHeader(name);
   const key = keys.find(k => normalizeHeader(k) === wanted);
   if (key !== undefined) return clean(row[key]);
 }

 return "";
}

function makeId(text, index) {
 const raw = text + "|" + index;
 let hash = 0;

 for (let i = 0; i < raw.length; i++) {
   hash = ((hash << 5) - hash) + raw.charCodeAt(i);
   hash |= 0;
 }

 return "q_" + Math.abs(hash) + "_" + index;
}

function normalizeCorrect(value) {
 const v = clean(value).toLowerCase();

 const map = {
   "a": "A",
   "b": "B",
   "c": "C",
   "d": "D",
   "1": "A",
   "2": "B",
   "3": "C",
   "4": "D",
   "optiona": "A",
   "optionb": "B",
   "optionc": "C",
   "optiond": "D",
   "option 1": "A",
   "option 2": "B",
   "option 3": "C",
   "option 4": "D"
 };

 return map[v.replace(/\s+/g, "")] || map[v] || "";
}

function normalizeQuestion(row, index, sheetName) {
 const question = getField(row, [
   "Question", "પ્રશ્ન", "questiontext", "question text"
 ]);

 const options = [
   getField(row, ["Option A", "A", "વિકલ્પ A", "વિકલ્પ 1"]),
   getField(row, ["Option B", "B", "વિકલ્પ B", "વિકલ્પ 2"]),
   getField(row, ["Option C", "C", "વિકલ્પ C", "વિકલ્પ 3"]),
   getField(row, ["Option D", "D", "વિકલ્પ D", "વિકલ્પ 4"])
 ];

 const correct = normalizeCorrect(getField(row, [
   "Correct Answer", "Answer", "Correct", "જવાબ", "સાચો જવાબ"
 ]));

 const explanation = getField(row, [
   "Explanation", "સમજૂતી", "Solution", "વ્યાખ્યા"
 ]);

 const hint = getField(row, ["Hint", "સંકેત"]);
 const subject = getField(row, ["Subject", "વિષય"]) || sheetName;
 const suppliedId = getField(row, ["ID", "Question ID", "પ્રશ્ન ID"]);

 if (!question || options.some(x => !x)) return null;
 if (!["A", "B", "C", "D"].includes(correct)) return null;

 const uniqueText = [
   question,
   ...options,
   subject
 ].join("¦").toLowerCase();

 return {
   id: suppliedId || makeId(uniqueText, index),
   question,
   options,
   correct,
   explanation,
   hint,
   subject: subject || "General",
   originalIndex: index
 };
}

function parseRows(rows, sheetName) {
 const result = [];

 rows.forEach((row, index) => {
   const q = normalizeQuestion(row, index, sheetName);
   if (q) result.push(q);
 });

 return result;
}

function deduplicate(questions) {
 const seen = new Set();
 const result = [];

 for (const q of questions) {
   const key = q.id || [
     q.question,
     ...q.options
   ].join("¦").toLowerCase();

   if (seen.has(key)) continue;

   seen.add(key);
   result.push(q);
 }

 return result;
}

function shuffle(array) {
 const result = [...array];

 for (let i = result.length - 1; i > 0; i--) {
   const j = Math.floor(Math.random() * (i + 1));
   [result[i], result[j]] = [result[j], result[i]];
 }

 return result;
}

function setStatus(message, error = false) {
 els.loadStatus.textContent = message;
 els.loadStatus.style.color = error ? "#b42318" : "";
}

function updateSubjects() {
 const current = els.subject.value;

 els.subject.innerHTML = "";
 const all = document.createElement("option");
 all.value = "";
 all.textContent = "બધા Subjects";
 els.subject.appendChild(all);

 const subjects = [...new Set(
   state.allQuestions.map(q => q.subject).filter(Boolean)
 )].sort((a, b) => a.localeCompare(b));

 subjects.forEach(subject => {
   const option = document.createElement("option");
   option.value = subject;
   option.textContent = subject;
   els.subject.appendChild(option);
 });

 if (subjects.includes(current)) els.subject.value = current;
}

function showBankInfo() {
 const subjects = [...new Set(state.allQuestions.map(q => q.subject))];

 els.bankInfo.classList.remove("hidden");
 els.bankInfo.textContent =
   "Loaded questions: " + state.allQuestions.length +
   "\nSubjects: " + subjects.length +
   "\nSubject names: " + subjects.join(", ");

 els.startBtn.disabled = state.allQuestions.length === 0;
}

function selectScheduleSubject(subjectName) {
 let found = false;
 for (let i = 0; i < els.subject.options.length; i++) {
   if (els.subject.options[i].value.toLowerCase() === subjectName.toLowerCase()) {
     els.subject.selectedIndex = i;
     found = true;
     break;
   }
 }

 if (!found) {
   const option = document.createElement("option");
   option.value = subjectName;
   option.textContent = subjectName;
   els.subject.appendChild(option);
   els.subject.value = subjectName;
 }

 els.setupPanel.scrollIntoView({ behavior: 'smooth' });
}

async function loadExcel(file) {
 if (!window.XLSX) {
   throw new Error("Excel library load થઈ નથી. Internet connection તપાસો.");
 }

 const buffer = await file.arrayBuffer();
 const workbook = XLSX.read(buffer, { type: "array" });
 let questions = [];

 workbook.SheetNames.forEach(sheetName => {
   const sheet = workbook.Sheets[sheetName];
   const rows = XLSX.utils.sheet_to_json(sheet, {
     defval: "",
     raw: false
   });

   questions = questions.concat(parseRows(rows, sheetName));
 });

 return questions;
}

async function loadGoogleSheet(url) {
 if (!url) throw new Error("Google Sheet CSV URL આપો.");

 const response = await fetch(url);

 if (!response.ok) {
   throw new Error("Google Sheet વાંચી શકાઈ નથી. URL અને publishing તપાસો.");
 }

 const csv = await response.text();

 if (!window.XLSX) {
   throw new Error("CSV reader ઉપલબ્ધ નથી.");
 }

 const workbook = XLSX.read(csv, { type: "string" });
 const sheetName = workbook.SheetNames[0];
 const sheet = workbook.Sheets[sheetName];

 const rows = XLSX.utils.sheet_to_json(sheet, {
   defval: "",
   raw: false
 });

 return parseRows(rows, sheetName);
}

async function loadBank() {
 setStatus("Question Bank load થઈ રહી છે...");
 els.loadBtn.disabled = true;

 try {
   let questions = [];

   if (els.sourceType.value === "excel") {
     const file = els.excelFile.files[0];

     if (!file) {
       throw new Error("Excel file પસંદ કરો.");
     }

     questions = await loadExcel(file);
     state.sourceName = file.name;
   } else {
     questions = await loadGoogleSheet(els.googleUrl.value.trim());
     state.sourceName = "Google Sheet";
   }

   questions = deduplicate(questions);

   if (!questions.length) {
     throw new Error(
       "માન્ય પ્રશ્નો મળ્યા નથી.\n" +
       "Column names અને Correct Answer તપાસો."
     );
   }

   state.allQuestions = questions;
   state.loaded = true;

   updateSubjects();
   showBankInfo();

   setStatus(
     "સફળતા!\n" +
     questions.length +
     " unique questions load થયા.\n" +
     "Source: " + state.sourceName
   );
 } catch (error) {
   state.allQuestions = [];
   state.loaded = false;
   els.startBtn.disabled = true;
   els.bankInfo.classList.add("hidden");
   setStatus(error.message || "Unknown error", true);
 } finally {
   els.loadBtn.disabled = false;
 }
}

function getFilteredQuestions() {
 const subject = els.subject.value;

 if (!subject) return [...state.allQuestions];

 return state.allQuestions.filter(q => q.subject.toLowerCase() === subject.toLowerCase());
}

function startQuiz() {
 const available = getFilteredQuestions();
 const count = Math.floor(Number(els.count.value));

 if (!available.length) {
   alert("આ Subject માટે કોઈ પ્રશ્ન મળ્યો નથી. કૃપા કરીને ડેટાબેઝ/એક્સેલ ફાઇલ લોડ કરો.");
   return;
 }

 if (!Number.isFinite(count) || count < 1) {
   alert("Questions count યોગ્ય આપો.");
   return;
 }

 const finalCount = Math.min(count, available.length);

 let selected = els.mode.value === "random"
   ? shuffle(available).slice(0, finalCount)
   : available.slice(0, finalCount);

 state.quizQuestions = selected;
 state.answers = new Array(selected.length).fill(null);
 state.currentIndex = 0;
 state.started = true;

 els.setupPanel.classList.add("hidden");
 els.resultPanel.classList.add("hidden");
 els.quizPanel.classList.remove("hidden");

 renderQuestion();
}

function renderQuestion() {
 const q = state.quizQuestions[state.currentIndex];
 const saved = state.answers[state.currentIndex];

 if (!q) return;

 els.quizSubject.textContent = q.subject;
 els.quizCount.textContent =
   "Question " + (state.currentIndex + 1) +
   " / " + state.quizQuestions.length;

 els.progressBar.style.width =
   ((state.currentIndex + 1) / state.quizQuestions.length * 100) + "%";

 els.questionText.textContent = q.question;
 els.options.innerHTML = "";
 els.feedback.className = "feedback hidden";
 els.feedback.textContent = "";

 const labels = ["A", "B", "C", "D"];

 q.options.forEach((text, index) => {
   const letter = labels[index];
   const button = document.createElement("button");

   button.type = "button";
   button.className = "option";
   button.dataset.answer = letter;

   const badge = document.createElement("span");
   badge.className = "letter";
   badge.textContent = letter;

   const content = document.createElement("span");
   content.textContent = text;

   button.appendChild(badge);
   button.appendChild(content);

   button.addEventListener("click", () => selectAnswer(letter));
   els.options.appendChild(button);
 });

 if (saved) {
   applyAnswer(saved);
 }

 els.prevBtn.disabled = state.currentIndex === 0;
 els.nextBtn.disabled = !saved;
 els.nextBtn.textContent =
   state.currentIndex === state.quizQuestions.length - 1
     ? "Finish Quiz"
     : "Next";
}

function selectAnswer(answer) {
 if (state.answers[state.currentIndex]) return;

 state.answers[state.currentIndex] = answer;
 applyAnswer(answer);
}

function applyAnswer(answer) {
 const q = state.quizQuestions[state.currentIndex];
 const isCorrect = answer === q.correct;

 [...els.options.children].forEach(button => {
   const letter = button.dataset.answer;
   button.disabled = true;

   if (letter === answer) {
     button.classList.add("selected");
   }

   if (letter === q.correct) {
     button.classList.add("correct");
   }

   if (letter === answer && !isCorrect) {
     button.classList.add("wrong");
   }
 });

 els.feedback.classList.remove("hidden");
 els.feedback.classList.add(isCorrect ? "correct" : "wrong");

 let message = isCorrect
   ? "સાચો જવાબ પસંદ કર્યો છે."
   : "પસંદ કરેલો જવાબ સાચો નથી.";

 if (q.explanation) {
   message += "\n\nસમજૂતી:\n" + q.explanation;
 }

 if (q.hint) {
   message += "\n\nસંકેત:\n" + q.hint;
 }

 els.feedback.textContent = message;
 els.nextBtn.disabled = false;
}

function goNext() {
 if (!state.answers[state.currentIndex]) return;

 if (state.currentIndex === state.quizQuestions.length - 1) {
   finishQuiz();
   return;
 }

 state.currentIndex++;
 renderQuestion();
}

function goPrevious() {
 if (state.currentIndex === 0) return;

 state.currentIndex--;
 renderQuestion();
}

function finishQuiz() {
 const total = state.quizQuestions.length;
 const correct = state.answers.filter(
   (answer, i) => answer === state.quizQuestions[i].correct
 ).length;

 const wrong = state.answers.filter(
   (answer, i) => answer && answer !== state.quizQuestions[i].correct
 ).length;

 const score = total ? Math.round(correct / total * 100) : 0;

 els.totalStat.textContent = total;
 els.correctStat.textContent = correct;
 els.wrongStat.textContent = wrong;
 els.scoreStat.textContent = score + "%";

 els.resultMessage.textContent =
   "તમારો Score: " + score + "%\n" +
   "સાચા જવાબ: " + correct + " / " + total;

 els.reviewList.innerHTML = "";

 state.quizQuestions.forEach((q, i) => {
   const userAnswer = state.answers[i];
   const item = document.createElement("div");
   item.className = "review-item";

   const title = document.createElement("strong");
   title.textContent = (i + 1) + ". " + q.question;

   const result = document.createElement("div");
   result.textContent =
     "તમારો જવાબ: " + (userAnswer || "Not Attempted") +
     "\nસાચો જવાબ: " + q.correct;

   const explanation = document.createElement("div");
   explanation.className = "small";
   explanation.textContent = q.explanation || "સમજૂતી ઉપલબ્ધ નથી.";

   item.appendChild(title);
   item.appendChild(result);
   item.appendChild(explanation);

   els.reviewList.appendChild(item);
 });

 els.quizPanel.classList.add("hidden");
 els.resultPanel.classList.remove("hidden");
}

function restartQuiz() {
 startQuiz();
}

function backToSetup() {
 state.started = false;
 els.quizPanel.classList.add("hidden");
 els.resultPanel.classList.add("hidden");
 els.setupPanel.classList.remove("hidden");
}

}
